% Model figure (two column, two panels):
%   (A) graphical model -- the accelerating accumulator with its three
%       measurement models (vocabulary / CDI items, input / recordings,
%       processing / looking-while-listening trials) and the structural links
%       that carry input and processing into efficiency (xi) and acceleration
%       (kappa); drawn in TikZ below.
%   (B) trajectory demo -- one child, then the same child with +1 SD of
%       efficiency (parallel shift) or +1 SD of acceleration (pivots on a0);
%       out/model_trajectories.pdf, rendered by fig_model_trajectories.R
%       (run it first; render_all.sh does).
% The TikZ diagram is built in a savebox and placed beside the included PDF;
% the panel letters (10 pt bold, like the R figures' tags) are added here.
% Notation follows the manuscript's Model section:
%   xi_i    = mu_xi    + d_i        + beta_w  w_i + alpha_i
%   kappa_i = mu_kappa + beta_d d_i + beta'_w w_i + zeta_i
%   theta_i(t) = xi_i + kappa_i log(t / a_0)
%   y_ijt ~ Bernoulli(logit^{-1}(theta_i(t) + log H - delta_j))   (CDI item j)
%   r_iv  ~ N(mu_{r,s} + d_i, sigma_m[instrument])                 (recording v)
%   q_in  ~ N(tau_s + w_i + psi log(t_n / a_0), sigma_q)           (LWL trial n)
% Shaded = observed; circles = latent; rounded boxes = deterministic;
% dashed circle = fixed input; small squares = study / instrument constants.
% Colours are the paper palette (paper/figures/_common.R): input green,
% processing purple, efficiency blue, acceleration vermillion; vocabulary grey.
% Each panel is 8.7 cm wide (PNAS single column) and the figure 17.8 cm
% (double column), so type is printed at size: node labels 8 pt
% (\footnotesize), small nodes / edge labels / plate labels 7 pt
% (\scriptsize), matching the R figures. Text is Helvetica (helvet).
% Compile: pdflatex fig_graphical_model.tex  (render_all.sh does this)
\documentclass[border=2pt]{standalone}
\usepackage{tikz}
\usepackage{amsmath}
\usepackage{graphicx}
\usepackage[scaled=0.92]{helvet}
\renewcommand{\familydefault}{\sfdefault}
\usetikzlibrary{positioning,calc,fit,arrows.meta,backgrounds}
\definecolor{eff}{HTML}{0072B2}
\definecolor{acc}{HTML}{D55E00}
\definecolor{inp}{HTML}{009E73}
\definecolor{proc}{HTML}{AA3377}
\tikzset{
  every node/.style={font=\footnotesize},
  lat/.style ={circle, draw=black, fill=white,    minimum size=0.68cm, inner sep=1pt},
  obs/.style ={circle, draw=black, fill=black!14, minimum size=0.68cm, inner sep=1pt},
  res/.style ={circle, draw=black!55, fill=white, minimum size=0.46cm, inner sep=0pt, font=\scriptsize},
  det/.style ={rectangle, rounded corners=2.5pt, draw=black!70, fill=black!5, minimum height=0.6cm, inner sep=3pt},
  fix/.style ={circle, draw=black!55, dashed, fill=white, minimum size=0.48cm, inner sep=1pt, font=\scriptsize},
  stu/.style ={rectangle, draw=black!55, fill=white, minimum size=0.46cm, inner sep=2pt, font=\scriptsize},
  pl/.style  ={draw=black!50, rounded corners=4pt, inner sep=5pt},
  pll/.style ={font=\scriptsize\itshape, black!55, inner sep=1pt},
  e/.style   ={-{Latex[length=1.5mm]}, draw=black!75},
  el/.style  ={font=\scriptsize, inner sep=1pt, fill=white},
  hdr/.style ={font=\scriptsize\bfseries, black!60},
}
