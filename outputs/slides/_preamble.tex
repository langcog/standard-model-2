% Compact tables, gentle row spacing

% Tighten paragraph spacing. Quarto/Pandoc emits blank lines between
% every Markdown block, which LaTeX turns into \parskip vertical space.
% Default \parskip in Beamer is ~6pt, which on a content-heavy slide
% with 4-5 paragraphs adds ~30pt of slack that pushes the last paragraph
% past the frame boundary. Setting to a smaller (but not zero) value
% recovers most of that without making the slide feel cramped.
\setlength{\parskip}{2pt plus 1pt minus 1pt}
\setlength{\partopsep}{0pt}
\setlength{\topsep}{2pt plus 1pt minus 1pt}
